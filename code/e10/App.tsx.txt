// Full Stack Challenge: aquí recuperas el patrón CRUD completo.
// La pantalla de referencia permite consultar y marcar destinos como visitados.
import { useEffect,useState } from 'react';
import { Button,FlatList,SafeAreaView,Text,View } from 'react-native';
type Destino={id:number;ciudad:string;pais:string;descripcion:string;prioridad:string;visitado:boolean};
const API_URL='http://TU_IP_LOCAL:3000';
export default function App(){
 const [destinos,setDestinos]=useState<Destino[]>([]);
 async function cargar(){const r=await fetch(API_URL+'/destinos');setDestinos(await r.json());}
 useEffect(()=>{cargar();},[]);
 async function visitar(id:number){await fetch(API_URL+'/destinos/'+id,{method:'PATCH',headers:{'Content-Type':'application/json'},body:JSON.stringify({visitado:true})});cargar();}
 return <SafeAreaView><Text>🌍 Wander</Text><FlatList data={destinos} keyExtractor={x=>String(x.id)}
 renderItem={({item})=><View><Text>{item.ciudad} · {item.pais}</Text><Text>{item.prioridad} · {item.visitado?'Visitado':'Pendiente'}</Text><Button title="Marcar visitado" onPress={()=>visitar(item.id)}/></View>}/></SafeAreaView>;
}
